```python
# app/config/provider_config.py

import yaml
import os
from typing import Dict, Any, List, Optional
from pydantic import BaseModel, validator

class ProviderConfig(BaseModel):
    """服务商配置基类"""
    provider_id: str
    provider_name: str
    base_url: str
    auth_type: str
    enabled: bool = True
    priority: int = 1
    
    @validator('base_url')
    def validate_base_url(cls, v):
        if not v.startswith(('http://', 'https://')):
            raise ValueError('Base URL must start with http:// or https://')
        return v

class ProviderManager:
    """服务商配置管理器"""
    
    def __init__(self, config_file: str = "config/providers.yaml"):
        self.config_file = config_file
        self.providers: Dict[str, Dict] = {}
        self.load_config()
    
    def load_config(self):
        """加载配置文件"""
        if not os.path.exists(self.config_file):
            raise FileNotFoundError(f"Provider config file not found: {self.config_file}")
        
        with open(self.config_file, 'r', encoding='utf-8') as f:
            config_data = yaml.safe_load(f)
        
        # 合并所有服务商配置
        self.providers.update(config_data.get('first_party', {}))
        self.providers.update(config_data.get('cloud_gateways', {}))
        self.providers.update(config_data.get('third_party', {}))
        
        # 替换环境变量
        self._replace_environment_variables()
    
    def _replace_environment_variables(self):
        """替换配置中的环境变量"""
        for provider_id, config in self.providers.items():
            for key, value in config.items():
                if isinstance(value, str) and value.startswith('${') and value.endswith('}'):
                    env_var = value[2:-1]
                    config[key] = os.getenv(env_var, '')
    
    def get_provider(self, provider_id: str) -> Optional[Dict]:
        """获取指定服务商配置"""
        return self.providers.get(provider_id)
    
    def get_enabled_providers(self) -> List[Dict]:
        """获取所有启用的服务商"""
        return [
            provider for provider in self.providers.values() 
            if provider.get('enabled', True)
        ]
    
    def get_providers_by_priority(self) -> List[Dict]:
        """按优先级排序获取服务商"""
        enabled_providers = self.get_enabled_providers()
        return sorted(enabled_providers, key=lambda x: x.get('priority', 999))
    
    def reload_config(self):
        """重新加载配置"""
        self.load_config()
```

```python
# app/adapters/base/unified_adapter.py

from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional, AsyncGenerator
from app.adapters.base.protocol_types import ProtocolType

class UnifiedModelAdapter(ABC):
    """统一模型适配器基类"""
    
    def __init__(self, provider_config: Dict, protocol: ProtocolType):
        self.provider_config = provider_config
        self.protocol = protocol
        self.client = None
        self._initialize_client()
    
    @abstractmethod
    def _initialize_client(self):
        """初始化协议客户端"""
        pass
    
    @abstractmethod
    async def chat_completion(self, messages: List[Dict], **kwargs) -> Dict:
        """聊天补全接口"""
        pass
    
    @abstractmethod
    async def stream_chat_completion(self, messages: List[Dict], **kwargs) -> AsyncGenerator:
        """流式聊天补全接口"""
        pass
    
    @abstractmethod
    def convert_to_standard_format(self, response: Dict) -> Dict:
        """转换为标准格式"""
        pass
    
    @abstractmethod
    def handle_error(self, error: Exception) -> Dict:
        """错误处理"""
        pass
    
    def get_protocol_info(self) -> Dict:
        """获取协议信息"""
        return {
            "protocol_type": self.protocol.value,
            "provider_id": self.provider_config['provider_id'],
            "provider_name": self.provider_config['provider_name'],
            "supported_features": self.provider_config.get('features', [])
        }
    
    def is_healthy(self) -> bool:
        """检查服务商健康状态"""
        # 基础健康检查实现
        return self.client is not None
```

```python
# app/adapters/protocols/openai_adapter.py

import openai
from typing import List, Dict, Any, AsyncGenerator
from app.adapters.base.unified_adapter import UnifiedModelAdapter
from app.adapters.base.protocol_types import ProtocolType

class OpenAIProtocolAdapter(UnifiedModelAdapter):
    """OpenAI协议适配器"""
    
    def __init__(self, provider_config: Dict):
        super().__init__(provider_config, ProtocolType.OPENAI)
    
    def _initialize_client(self):
        """初始化OpenAI客户端"""
        api_key = self.provider_config.get('api_key')
        base_url = self.provider_config['base_url']
        timeout = self.provider_config.get('timeout', 30)
        
        if not api_key:
            raise ValueError("OpenAI API key is required")
        
        self.client = openai.AsyncOpenAI(
            api_key=api_key,
            base_url=base_url,
            timeout=timeout
        )
    
    async def chat_completion(self, messages: List[Dict], **kwargs) -> Dict:
        """聊天补全"""
        try:
            response = await self.client.chat.completions.create(
                model=kwargs.get('model', 'gpt-3.5-turbo'),
                messages=messages,
                **kwargs
            )
            return self.convert_to_standard_format(response.dict())
        except Exception as e:
            return self.handle_error(e)
    
    async def stream_chat_completion(self, messages: List[Dict], **kwargs) -> AsyncGenerator:
        """流式聊天补全"""
        try:
            stream = await self.client.chat.completions.create(
                model=kwargs.get('model', 'gpt-3.5-turbo'),
                messages=messages,
                stream=True,
                **kwargs
            )
            
            async for chunk in stream:
                yield self.convert_to_standard_format(chunk.dict())
        except Exception as e:
            yield self.handle_error(e)
    
    def convert_to_standard_format(self, response: Dict) -> Dict:
        """转换为标准格式"""
        return {
            "id": response.get("id"),
            "object": "chat.completion",
            "created": response.get("created"),
            "model": response.get("model"),
            "choices": [{
                "index": 0,
                "message": {
                    "role": "assistant",
                    "content": response.get("choices", [{}])[0].get("message", {}).get("content", "")
                },
                "finish_reason": "stop"
            }]
        }
    
    def handle_error(self, error: Exception) -> Dict:
        """错误处理"""
        return {
            "error": {
                "type": type(error).__name__,
                "message": str(error),
                "provider": self.provider_config['provider_id']
            }
        }
```

```python
# app/adapters/adapter_factory.py

from typing import Dict, Optional
from app.adapters.base.unified_adapter import UnifiedModelAdapter
from app.adapters.base.protocol_types import ProtocolType
from app.adapters.protocols.openai_adapter import OpenAIProtocolAdapter
from app.adapters.protocols.anthropic_adapter import AnthropicProtocolAdapter

class UnifiedAdapterFactory:
    """统一适配器工厂"""
    
    def __init__(self):
        self._adapters: Dict[str, UnifiedModelAdapter] = {}
    
    def create_adapter(self, provider_config: Dict) -> Optional[UnifiedModelAdapter]:
        """创建协议适配器"""
        provider_id = provider_config['provider_id']
        
        # 检查缓存
        if provider_id in self._adapters:
            return self._adapters[provider_id]
        
        # 确定协议类型
        protocol = self._determine_protocol(provider_config)
        
        # 创建适配器
        adapter = self._create_adapter_by_protocol(provider_config, protocol)
        
        if adapter:
            self._adapters[provider_id] = adapter
        
        return adapter
    
    def _determine_protocol(self, provider_config: Dict) -> ProtocolType:
        """确定协议类型"""
        supported_protocols = provider_config.get('supported_protocols', [])
        
        # 优先级：OpenAI > Anthropic > vLLM
        for protocol in [ProtocolType.OPENAI, ProtocolType.ANTHROPIC, ProtocolType.VLLM]:
            if protocol.value in supported_protocols:
                return protocol
        
        # 检查网关协议
        gateway_protocol = provider_config.get('gateway_protocol')
        if gateway_protocol:
            return ProtocolType(gateway_protocol)
        
        # 默认使用OpenAI协议
        return ProtocolType.OPENAI
    
    def _create_adapter_by_protocol(self, provider_config: Dict, protocol: ProtocolType) -> Optional[UnifiedModelAdapter]:
        """根据协议创建适配器"""
        if protocol == ProtocolType.OPENAI:
            return OpenAIProtocolAdapter(provider_config)
        elif protocol == ProtocolType.ANTHROPIC:
            return AnthropicProtocolAdapter(provider_config)
        # 添加其他协议适配器...
        
        return None
    
    def clear_cache(self):
        """清空适配器缓存"""
        self._adapters.clear()
```

```python
# app/services/usage_tracker.py

import time
from datetime import datetime, timedelta
from typing import Dict, List, Optional
from dataclasses import dataclass

@dataclass
class UsageRecord:
    """使用量记录"""
    provider_id: str
    timestamp: float
    tokens_used: int
    request_type: str
    success: bool

class UsageTracker:
    """使用量跟踪器"""
    
    def __init__(self, retention_days: int = 30):
        self.retention_days = retention_days
        self.usage_records: List[UsageRecord] = []
    
    def record_usage(self, provider_id: str, tokens_used: int, request_type: str, success: bool = True):
        """记录使用量"""
        record = UsageRecord(
            provider_id=provider_id,
            timestamp=time.time(),
            tokens_used=tokens_used,
            request_type=request_type,
            success=success
        )
        self.usage_records.append(record)
        self._cleanup_old_records()
    
    def get_daily_usage(self, provider_id: str) -> int:
        """获取今日使用量"""
        today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
        today_timestamp = today.timestamp()
        
        daily_usage = sum(
            record.tokens_used for record in self.usage_records
            if record.provider_id == provider_id and record.timestamp >= today_timestamp
        )
        return daily_usage
    
    def get_monthly_usage(self, provider_id: str) -> int:
        """获取本月使用量"""
        first_day_of_month = datetime.now().replace(day=1, hour=0, minute=0, second=0, microsecond=0)
        month_timestamp = first_day_of_month.timestamp()
        
        monthly_usage = sum(
            record.tokens_used for record in self.usage_records
            if record.provider_id == provider_id and record.timestamp >= month_timestamp
        )
        return monthly_usage
    
    def get_daily_request_count(self, provider_id: str) -> int:
        """获取今日请求次数"""
        today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
        today_timestamp = today.timestamp()
        
        daily_requests = sum(
            1 for record in self.usage_records
            if record.provider_id == provider_id and record.timestamp >= today_timestamp
        )
        return daily_requests
    
    def _cleanup_old_records(self):
        """清理过期记录"""
        cutoff_time = time.time() - (self.retention_days * 24 * 60 * 60)
        self.usage_records = [
            record for record in self.usage_records
            if record.timestamp >= cutoff_time
        ]
```

### 4.2 实现免费服务商路由器

```python
# app/services/free_provider_router.py

from typing import Dict, List, Optional
from app.config.provider_config import ProviderManager
from app.services.usage_tracker import UsageTracker

class FreeProviderRouter:
    """免费服务商智能路由器"""
    
    def __init__(self, provider_manager: ProviderManager, usage_tracker: UsageTracker):
        self.provider_manager = provider_manager
        self.usage_tracker = usage_tracker
    
    def get_available_free_providers(self, requirements: Dict) -> List[Dict]:
        """获取可用的免费服务商"""
        free_providers = self._get_all_free_providers()
        
        if not free_providers:
            return []
        
        # 检查免费额度和特性支持
        available_providers = []
        for provider in free_providers:
            if self._check_free_quota(provider) and self._check_feature_support(provider, requirements):
                available_providers.append(provider)
        
        return available_providers
    
    def _get_all_free_providers(self) -> List[Dict]:
        """获取所有免费服务商"""
        all_providers = self.provider_manager.providers
        return [
            provider for provider in all_providers.values()
            if provider.get('pricing_tier') == 'free' and provider.get('enabled', True)
        ]
    
    def _check_free_quota(self, provider: Dict) -> bool:
        """检查免费额度"""
        provider_id = provider['provider_id']
        free_limits = provider.get('free_limits', {})
        
        # 检查每日请求限制
        daily_requests = self.usage_tracker.get_daily_request_count(provider_id)
        if daily_requests >= free_limits.get('daily_requests', 0):
            return False
        
        # 检查每月请求限制
        monthly_usage = self.usage_tracker.get_monthly_usage(provider_id)
        if monthly_usage >= free_limits.get('monthly_requests', 0):
            return False
        
        return True
    
    def _check_feature_support(self, provider: Dict, requirements: Dict) -> bool:
        """检查特性支持"""
        required_features = requirements.get('features', [])
        provider_features = provider.get('features', [])
        
        # 如果不需要特定特性，则所有服务商都支持
        if not required_features:
            return True
        
        # 检查是否支持所有必需特性
        return all(feature in provider_features for feature in required_features)
    
    def get_free_provider_priority(self, provider: Dict) -> float:
        """计算免费服务商优先级"""
        provider_id = provider['provider_id']
        free_limits = provider.get('free_limits', {})
        
        # 计算剩余额度比例
        daily_requests = self.usage_tracker.get_daily_request_count(provider_id)
        daily_limit = free_limits.get('daily_requests', 1)
        daily_ratio = (daily_limit - daily_requests) / daily_limit
        
        monthly_usage = self.usage_tracker.get_monthly_usage(provider_id)
        monthly_limit = free_limits.get('monthly_requests', 1)
        monthly_ratio = (monthly_limit - monthly_usage) / monthly_limit
        
        # 综合优先级（额度越充足，优先级越高）
        priority = (daily_ratio + monthly_ratio) / 2
        return max(0.0, min(1.0, priority))
```

### 4.3 实现聚合管理器（优化版）

```python
# app/services/provider_aggregator.py

from typing import Dict, List, Optional, Any
from dataclasses import dataclass
from app.adapters.base.unified_adapter import UnifiedModelAdapter
from app.adapters.adapter_factory import UnifiedAdapterFactory
from app.config.provider_config import ProviderManager
from app.services.usage_tracker import UsageTracker
from app.services.free_provider_router import FreeProviderRouter

@dataclass
class ProviderHealth:
    """服务商健康状态"""
    provider_id: str
    status: str  # "healthy", "degraded", "unhealthy"
    response_time: float
    success_rate: float
    last_check: float

class ProviderAggregator:
    """服务商聚合管理器"""
    
    def __init__(self):
        self.provider_manager = ProviderManager()
        self.adapter_factory = UnifiedAdapterFactory()
        self.usage_tracker = UsageTracker()
        self.free_router = FreeProviderRouter(self.provider_manager, self.usage_tracker)
        self.adapters: Dict[str, UnifiedModelAdapter] = {}
        self.health_status: Dict[str, ProviderHealth] = {}
        self._initialize_adapters()
    
    def _initialize_adapters(self):
        """初始化所有适配器"""
        for provider_id, config in self.provider_manager.providers.items():
            if config.get('enabled', True):
                adapter = self.adapter_factory.create_adapter(config)
                if adapter:
                    self.adapters[provider_id] = adapter
                    # 初始化健康状态
                    self.health_status[provider_id] = ProviderHealth(
                        provider_id=provider_id,
                        status="healthy",
                        response_time=0.0,
                        success_rate=1.0,
                        last_check=0.0
                    )
    
    async def get_best_provider(self, requirements: Dict) -> Optional[UnifiedModelAdapter]:
        """获取最优服务商"""
        # 优先尝试免费服务商
        free_providers = self.free_router.get_available_free_providers(requirements)
        
        if free_providers:
            # 选择优先级最高的免费服务商
            best_free_provider = max(
                free_providers, 
                key=lambda p: self.free_router.get_free_provider_priority(p)
            )
            return self.adapters.get(best_free_provider['provider_id'])
        
        # 如果没有可用的免费服务商，使用付费服务商
        candidates = self._filter_providers(requirements)
        
        if not candidates:
            return None
        
        # 评分算法
        scored_providers = []
        for provider_id in candidates:
            score = self._calculate_provider_score(provider_id, requirements)
            scored_providers.append((provider_id, score))
        
        # 选择评分最高的服务商
        best_provider_id = max(scored_providers, key=lambda x: x[1])[0]
        return self.adapters.get(best_provider_id)
    
    def _filter_providers(self, requirements: Dict) -> List[str]:
        """过滤符合条件的服务商"""
        candidates = []
        
        for provider_id, adapter in self.adapters.items():
            config = self.provider_manager.get_provider(provider_id)
            
            # 跳过免费服务商（已经在免费路由中处理）
            if config.get('pricing_tier') == 'free':
                continue
            
            # 检查模型支持
            required_model = requirements.get('model')
            if required_model and not self._check_model_support(config, required_model):
                continue
            
            # 检查特性支持
            if not self._check_feature_support(config, requirements.get('features', [])):
                continue
            
            # 检查健康状态
            health = self.health_status.get(provider_id)
            if health and health.status == "unhealthy":
                continue
            
            candidates.append(provider_id)
        
        return candidates
    
    def _check_model_support(self, config: Dict, model: str) -> bool:
        """检查模型支持"""
        supported_models = config.get('supported_models', [])
        
        # 如果没有指定支持的模型，则默认支持所有模型
        if not supported_models:
            return True
        
        return model in supported_models
    
    def _check_feature_support(self, config: Dict, required_features: List[str]) -> bool:
        """检查特性支持"""
        provider_features = config.get('features', [])
        return all(feature in provider_features for feature in required_features)
    
    def _calculate_provider_score(self, provider_id: str, requirements: Dict) -> float:
        """计算服务商评分"""
        config = self.provider_manager.get_provider(provider_id)
        health = self.health_status.get(provider_id)
        
        # 成本因子 (40%权重)
        cost_score = self._calculate_cost_score(config, requirements)
        
        # 性能因子 (25%权重)
        performance_score = self._calculate_performance_score(health)
        
        # 可靠性因子 (25%权重)
        reliability_score = self._calculate_reliability_score(config, health)
        
        # 免费优先级因子 (10%权重)
        free_priority_score = self._calculate_free_priority_score(provider_id)
        
        # 综合评分
        total_score = (
            0.4 * cost_score + 
            0.25 * performance_score + 
            0.25 * reliability_score +
            0.1 * free_priority_score
        )
        
        return total_score
    
    def _calculate_free_priority_score(self, provider_id: str) -> float:
        """计算免费优先级评分"""
        provider = self.provider_manager.get_provider(provider_id)
        
        if provider.get('pricing_tier') == 'free':
            # 免费服务商获得额外加分
            free_limits = provider.get('free_limits', {})
            
            # 基于剩余额度计算优先级
            daily_requests = self.usage_tracker.get_daily_request_count(provider_id)
            daily_limit = free_limits.get('daily_requests', 1)
            daily_ratio = (daily_limit - daily_requests) / daily_limit
            
            monthly_usage = self.usage_tracker.get_monthly_usage(provider_id)
            monthly_limit = free_limits.get('monthly_requests', 1)
            monthly_ratio = (monthly_limit - monthly_usage) / monthly_limit
            
            return (daily_ratio + monthly_ratio) / 2
        
        return 0.0
```

```python
from app.services.provider_aggregator import ProviderAggregator

class EnhancedRoutingService(RoutingService):

        self.provider_aggregator = ProviderAggregator()
    
    async def select_optimal_provider(self, query: str, intent: str, 
                                    complexity: float, requirements: Dict) -> Optional[UnifiedModelAdapter]:
        """选择最优服务商"""
        
        # 构建需求参数
        provider_requirements = {
            "model": requirements.get('model'),
            "features": self._get_required_features(intent, complexity),
            "max_cost": requirements.get('max_cost', 0.1),
            "min_quality": requirements.get('min_quality', 0.8)

        return await self.provider_aggregator.get_best_provider(provider_requirements)

        if intent == "code":
            features.append("function_calling")
        
        if complexity > 0.7:
            features.append("long_context")
        
        return features
```
