```python
# app/services/cost_optimization_service.py

from typing import List, Dict, Optional
from dataclasses import dataclass
from app.models import ModelStatus
from app.utils.telemetry import get_tracer

@dataclass
class CostOptimizationConfig:
    """成本优化配置"""
    cost_weight: float = 0.4
    performance_weight: float = 0.3
    capability_weight: float = 0.2
    quota_weight: float = 0.1
    min_performance_threshold: float = 0.7  # 最低性能要求
    max_cost_increase: float = 0.2  # 最大成本增加幅度

class CostOptimizationService:
    """成本优化服务"""
    
    def __init__(self, config: Optional[CostOptimizationConfig] = None):
        self.config = config or CostOptimizationConfig()
        self.tracer = get_tracer()
        
    def calculate_model_score(self, model: ModelStatus, intent: str, 
                            complexity: float, current_quota_usage: Dict) -> float:
        """计算模型综合评分"""
        
        with self.tracer.start_as_current_span("calculate_model_score", attributes={
            "model_id": model.model_id,
            "intent": intent,
            "complexity": complexity
        }):
            
            # 成本因子
            cost_factor = self._calculate_cost_factor(model)
            
            # 性能因子
            performance_factor = self._calculate_performance_factor(model)
            
            # 能力匹配因子
            capability_factor = self._calculate_capability_factor(model, intent, complexity)
            
            # 配额健康度因子
            quota_factor = self._calculate_quota_factor(model, current_quota_usage)
            
            # 综合评分
            total_score = (
                self.config.cost_weight * cost_factor +
                self.config.performance_weight * performance_factor +
                self.config.capability_weight * capability_factor +
                self.config.quota_weight * quota_factor
            )
            
            return total_score
    
    def _calculate_cost_factor(self, model: ModelStatus) -> float:
        """计算成本因子"""
        # 获取实时价格
        current_price = self._get_real_time_price(model.model_id)
        
        # 归一化处理
        max_price = self._get_max_price_in_catalog()
        if max_price == 0:
            return 1.0
            
        normalized_price = current_price / max_price
        cost_factor = 1.0 - normalized_price
        
        return max(0.0, min(1.0, cost_factor))
    
    def _calculate_performance_factor(self, model: ModelStatus) -> float:
        """计算性能因子"""
        # 基于历史响应时间计算性能评分
        avg_response_time = self._get_avg_response_time(model.model_id)
        
        # 响应时间越短，性能评分越高
        if avg_response_time <= 1000:  # 1秒以内
            return 1.0
        elif avg_response_time <= 3000:  # 3秒以内
            return 0.8
        elif avg_response_time <= 5000:  # 5秒以内
            return 0.6
        else:
            return 0.4
    
    def _calculate_capability_factor(self, model: ModelStatus, intent: str, complexity: float) -> float:
        """计算能力匹配因子"""
        # 根据意图和复杂度评估模型能力匹配度
        
        # 简单任务可以使用低成本模型
        if complexity < 0.3:
            return 1.0  # 所有模型都适合简单任务
        
        # 复杂任务需要高质量模型
        if complexity > 0.7:
            if model.quality_tier == "large":
                return 1.0
            else:
                return 0.5
        
        # 中等复杂度任务
        return 0.8
    
    def _calculate_quota_factor(self, model: ModelStatus, quota_usage: Dict) -> float:
        """计算配额健康度因子"""
        model_quota = quota_usage.get(model.model_id, {})
        used = model_quota.get("used", 0)
        limit = model_quota.get("limit", float('inf'))
        
        if limit == float('inf') or limit == 0:
            return 1.0
        
        usage_ratio = used / limit
        
        # 使用率越低，评分越高
        if usage_ratio < 0.5:
            return 1.0
        elif usage_ratio < 0.8:
            return 0.7
        elif usage_ratio < 0.9:
            return 0.4
        else:
            return 0.1  # 接近配额上限，评分很低
```

```python
class EnhancedRoutingService(RoutingService):
    """增强的路由服务，集成成本优化"""
    
    def __init__(self, model_service: ModelService):
        super().__init__(model_service)
        self.cost_optimization_service = CostOptimizationService()
        self.quota_manager = QuotaManager()
    
    def get_optimal_model(self, query: str, intent: str, 
                         reasoning_level: Optional[str] = None, 
                         complexity: Optional[float] = None) -> ModelStatus:
        """获取最优模型，考虑成本优化"""
        
        # 计算复杂度（如果未提供）
        if complexity is None:
            complexity = self.get_intent_complexity(query)
        
        # 获取候选模型
        candidate_models = self._get_candidate_models(intent, reasoning_level)
        
        # 获取当前配额使用情况
        quota_usage = self.quota_manager.get_current_quota_usage()
        
        # 计算各模型评分
        scored_models = []
        for model in candidate_models:
            score = self.cost_optimization_service.calculate_model_score(
                model, intent, complexity, quota_usage
            )
            scored_models.append((model, score))
        
        # 选择评分最高的模型
        best_model, best_score = max(scored_models, key=lambda x: x[1])
        
        # 记录路由决策
        self._log_routing_decision(query, intent, complexity, 
                                 best_model.model_id, best_score)
        
        return best_model
```

```python
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler
from typing import Dict, List

class CostPredictionService:
    """成本预测服务"""
    
    def __init__(self):
        self.model = LinearRegression()
        self.scaler = StandardScaler()
        self.is_trained = False
    
    def train_model(self, historical_data: List[Dict]):
        """训练预测模型"""
        # 准备训练数据
        df = pd.DataFrame(historical_data)
        
        # 特征工程
        features = self._extract_features(df)
        targets = df['cost'].values
        
        # 数据标准化
        features_scaled = self.scaler.fit_transform(features)
        
        # 训练模型
        self.model.fit(features_scaled, targets)
        self.is_trained = True
    
    def predict_cost(self, features: Dict) -> float:
        """预测成本"""
        if not self.is_trained:
            return self._naive_prediction(features)
        
        # 特征处理
        feature_vector = self._prepare_feature_vector(features)
        feature_vector_scaled = self.scaler.transform([feature_vector])
        
        # 预测
        prediction = self.model.predict(feature_vector_scaled)[0]
        
        return max(0, prediction)  # 成本不能为负数
```
