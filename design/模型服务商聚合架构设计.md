# Aperture 模型服务商聚合架构设计

## 1. 架构概述

本架构设计旨在为Aperture提供灵活、可扩展的模型服务商聚合方案，支持多种类型的服务商和协议标准。基于对现有免费API资源的分析，本设计特别优化了免费和优惠政策的集成。

## 2. 服务商分类体系

### 2.1 第一方模型服务商
- **OpenAI**: GPT系列模型，支持OpenAI协议
- **Anthropic**: Claude系列模型，支持Anthropic协议  
- **Google**: Gemini系列模型，支持OpenAI协议
- **智谱AI**: GLM系列模型，支持OpenAI协议

### 2.2 云服务商网关
- **Azure AI**: 微软云AI服务，支持OpenAI协议
- **阿里云通义千问**: 阿里云AI服务，支持OpenAI协议

### 2.3 第三方模型服务商
- **ModelScope**: 开源模型平台，支持OpenAI/Anthropic协议
- **OpenRouter**: 模型聚合平台，支持OpenAI协议，提供免费模型路由
- **ChatAnywhere**: 提供OpenAI兼容API，支持多种模型

### 2.4 免费服务商（新增）
- **OpenRouter Free**: 免费模型路由，自动选择可用免费模型
- **Google AI Studio**: 提供免费Gemini模型访问
- **NVIDIA NIM**: 免费推理服务
- **Mistral La Plateforme**: 免费Mistral模型
- **HuggingFace Inference**: 免费开源模型推理
- **Cloudflare Workers AI**: 免费AI推理服务
- **Vercel AI Gateway**: 免费AI网关服务

### 2.2 协议标准定义

```python
# app/config/provider_config.py

import yaml
from typing import Dict, Any, List
from pydantic import BaseModel

class ProviderConfig(BaseModel):
    """服务商配置基类"""
    provider_id: str
    provider_name: str
    base_url: str
    auth_type: str
    enabled: bool = True
    priority: int = 1  # 优先级，数字越小优先级越高

class ProviderManager:
    """服务商配置管理器"""
    
    def __init__(self, config_file: str = "providers.yaml"):
        self.config_file = config_file
        self.providers: Dict[str, Dict] = {}
        self.load_config()
    
    def load_config(self):
        """加载配置文件"""
        with open(self.config_file, 'r', encoding='utf-8') as f:
            config_data = yaml.safe_load(f)
        
        # 合并所有服务商配置
        self.providers.update(config_data.get('first_party', {}))
        self.providers.update(config_data.get('cloud_gateways', {}))
        self.providers.update(config_data.get('third_party', {}))
    
    def get_provider(self, provider_id: str) -> Optional[Dict]:
        """获取指定服务商配置"""
        return self.providers.get(provider_id)
    
    def get_enabled_providers(self) -> List[Dict]:
        """获取所有启用的服务商"""
        return [
            provider for provider in self.providers.values() 
            if provider.get('enabled', True)
        ]
    
    def reload_config(self):
        """重新加载配置"""
        self.load_config()
```

### 3.2 协议适配器架构

#### 3.2.1 统一适配器接口
```python
# app/adapters/base/unified_adapter.py

from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional
from enum import Enum

class ProtocolType(Enum):
    """协议类型枚举"""
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    VLLM = "vllm"
    CUSTOM = "custom"

class UnifiedModelAdapter(ABC):
    """统一模型适配器基类"""
    
    def __init__(self, provider_config: Dict, protocol: ProtocolType):
        self.provider_config = provider_config
        self.protocol = protocol
        self.client = self._create_client()
    
    @abstractmethod
    def _create_client(self):
        """创建协议客户端"""
        pass
    
    @abstractmethod
    async def chat_completion(self, messages: List[Dict], **kwargs) -> Dict:
        """聊天补全接口"""
        pass
    
    @abstractmethod
    async def stream_chat_completion(self, messages: List[Dict], **kwargs):
        """流式聊天补全接口"""
        pass
    
    @abstractmethod
    def convert_to_standard_format(self, response: Dict) -> Dict:
        """转换为标准格式"""
        pass
    
    def get_protocol_info(self) -> Dict:
        """获取协议信息"""
        return {
            "protocol_type": self.protocol.value,
            "provider_id": self.provider_config['provider_id'],
            "supported_features": self.provider_config.get('features', [])
        }
```

#### 3.2.2 协议适配器实现
```python
# app/adapters/protocols/openai_adapter.py

import openai
from typing import List, Dict, Any, AsyncGenerator
from app.adapters.base.unified_adapter import UnifiedModelAdapter, ProtocolType

class OpenAIProtocolAdapter(UnifiedModelAdapter):
    """OpenAI协议适配器"""
    
    def __init__(self, provider_config: Dict):
        super().__init__(provider_config, ProtocolType.OPENAI)
    
    def _create_client(self):
        """创建OpenAI客户端"""
        return openai.AsyncOpenAI(
            api_key=self.provider_config.get('api_key'),
            base_url=self.provider_config['base_url'],
            timeout=self.provider_config.get('timeout', 30)
        )
    
    async def chat_completion(self, messages: List[Dict], **kwargs) -> Dict:
        """聊天补全"""
        try:
            response = await self.client.chat.completions.create(
                model=kwargs.get('model', 'gpt-3.5-turbo'),
                messages=messages,
                **kwargs
            )
            return self.convert_to_standard_format(response.dict())
        except Exception as e:
            return self._handle_error(e)
    
    async def stream_chat_completion(self, messages: List[Dict], **kwargs) -> AsyncGenerator:
        """流式聊天补全"""
        try:
            stream = await self.client.chat.completions.create(
                model=kwargs.get('model', 'gpt-3.5-turbo'),
                messages=messages,
                stream=True,
                **kwargs
            )
            
            async for chunk in stream:
                yield self.convert_to_standard_format(chunk.dict())
        except Exception as e:
            yield self._handle_error(e)
    
    def convert_to_standard_format(self, response: Dict) -> Dict:
        """转换为标准格式"""
        return {
            "id": response.get("id"),
            "object": "chat.completion",
            "created": response.get("created"),
            "model": response.get("model"),
            "choices": [{
                "index": 0,
                "message": {
                    "role": "assistant",
                    "content": response.get("choices", [{}])[0].get("message", {}).get("content", "")
                },
                "finish_reason": "stop"
            }]
        }
```

### 4.1 聚合管理器设计

```python
# app/services/provider_aggregator.py

from typing import Dict, List, Optional, Any
from dataclasses import dataclass
from app.adapters.base.unified_adapter import UnifiedModelAdapter, ProtocolType
from app.config.provider_config import ProviderManager

@dataclass
class ProviderHealth:
    """服务商健康状态"""
    provider_id: str
    status: Literal["healthy", "degraded", "unhealthy"]
    response_time: float  # 平均响应时间
    success_rate: float  # 成功率
    last_check: float  # 最后检查时间

class ProviderAggregator:
    """服务商聚合管理器"""
    
    def __init__(self):
        self.provider_manager = ProviderManager()
        self.adapters: Dict[str, UnifiedModelAdapter] = {}
        self.health_status: Dict[str, ProviderHealth] = {}
        self._initialize_adapters()
    
    def _initialize_adapters(self):
        """初始化所有适配器"""
        for provider_id, config in self.provider_manager.providers.items():
            if config.get('enabled', True):
                adapter = self._create_adapter(provider_id, config)
                if adapter:
                    self.adapters[provider_id] = adapter
    
    def _create_adapter(self, provider_id: str, config: Dict) -> Optional[UnifiedModelAdapter]:
        """创建协议适配器"""
        protocol = self._determine_protocol(config)
        
        if protocol == ProtocolType.OPENAI:
            from app.adapters.protocols.openai_adapter import OpenAIProtocolAdapter
            return OpenAIProtocolAdapter(config)
        elif protocol == ProtocolType.ANTHROPIC:
            from app.adapters.protocols.anthropic_adapter import AnthropicProtocolAdapter
            return AnthropicProtocolAdapter(config)
        elif protocol == ProtocolType.VLLM:
            from app.adapters.protocols.vllm_adapter import VLLMProtocolAdapter
            return VLLMProtocolAdapter(config)
        
        return None
    
    def _determine_protocol(self, config: Dict) -> ProtocolType:
        """确定协议类型"""
        supported_protocols = config.get('supported_protocols', [])
        
        # 优先级：OpenAI > Anthropic > vLLM
        for protocol in [ProtocolType.OPENAI, ProtocolType.ANTHROPIC, ProtocolType.VLLM]:
            if protocol.value in supported_protocols:
                return protocol
        
        return ProtocolType.OPENAI  # 默认使用OpenAI协议
    
    async def get_best_provider(self, requirements: Dict) -> Optional[UnifiedModelAdapter]:
        """获取最优服务商"""
        # 基于成本、性能、特性等因素选择最优服务商
        candidates = self._filter_providers(requirements)
        
        if not candidates:
            return None
        
        # 评分算法
        scored_providers = []
        for provider_id in candidates:
            score = self._calculate_provider_score(provider_id, requirements)
            scored_providers.append((provider_id, score))
        
        # 选择评分最高的服务商
        best_provider_id = max(scored_providers, key=lambda x: x[1])[0]
        return self.adapters.get(best_provider_id)
    
    def _filter_providers(self, requirements: Dict) -> List[str]:
        """过滤符合条件的服务商"""
        candidates = []
        
        for provider_id, adapter in self.adapters.items():
            config = self.provider_manager.get_provider(provider_id)
            
            # 检查特性支持
            if not self._check_feature_support(config, requirements.get('features', [])):
                continue
            
            # 检查健康状态
            health = self.health_status.get(provider_id)
            if health and health.status == "unhealthy":
                continue
            
            candidates.append(provider_id)
        
        return candidates
    
    def _check_feature_support(self, config: Dict, required_features: List[str]) -> bool:
        """检查特性支持"""
        provider_features = config.get('features', [])
        return all(feature in provider_features for feature in required_features)
    
    def _calculate_provider_score(self, provider_id: str, requirements: Dict) -> float:
        """计算服务商评分"""
        config = self.provider_manager.get_provider(provider_id)
        health = self.health_status.get(provider_id)
        
        # 成本因子 (40%权重)
        cost_score = self._calculate_cost_score(config, requirements)
        
        # 性能因子 (25%权重)
        performance_score = self._calculate_performance_score(health)
        
        # 可靠性因子 (25%权重)
        reliability_score = self._calculate_reliability_score(config, health)
        
        # 免费优先级因子 (10%权重)
        free_priority_score = self._calculate_free_priority_score(provider_id)
        
        # 综合评分
        total_score = (
            0.4 * cost_score + 
            0.25 * performance_score + 
            0.25 * reliability_score +
            0.1 * free_priority_score
        )
        
        return total_score
    
    def _calculate_free_priority_score(self, provider_id: str) -> float:
        """计算免费优先级评分"""
        provider = self.provider_manager.get_provider(provider_id)
        
        if provider.get('pricing_tier') == 'free':
            # 免费服务商获得额外加分
            free_limits = provider.get('free_limits', {})
            
            # 基于剩余额度计算优先级
            # 这里需要实现UsageTracker来跟踪使用情况
            # 暂时返回基础加分
            return 0.8
        
        return 0.0
```

## 5. 优惠政策和成本优化

### 5.1 优惠政策管理

```python
# app/services/promotion_manager.py

from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta

@dataclass
class PromotionPolicy:
    """优惠政策"""
    provider_id: str
    policy_type: Literal["free_tier", "volume_discount", "time_limited"]
    description: str
    start_time: datetime
    end_time: Optional[datetime]
    conditions: Dict  # 优惠条件
    benefits: Dict  # 优惠内容

class PromotionManager:
    """优惠政策管理器"""
    
    def __init__(self):
        self.policies: Dict[str, List[PromotionPolicy]] = {}
        self._load_promotion_policies()
    
    def _load_promotion_policies(self):
        """加载优惠政策"""
        # 从配置文件或数据库加载优惠政策
        self.policies = {
            "openai": [
                PromotionPolicy(
                    provider_id="openai",
                    policy_type="free_tier",
                    description="免费额度",
                    start_time=datetime.now(),
                    end_time=None,
                    conditions={"max_tokens": 1000},
                    benefits={"discount_rate": 1.0}  # 100%折扣
                )
            ],
            "anthropic": [
                PromotionPolicy(
                    provider_id="anthropic",
                    policy_type="volume_discount",
                    description="用量折扣",
                    start_time=datetime.now(),
                    end_time=datetime.now() + timedelta(days=30),
                    conditions={"min_tokens": 100000},
                    benefits={"discount_rate": 0.8}  # 20%折扣
                )
            ]
        }
    
    def get_applicable_policies(self, provider_id: str, usage: Dict) -> List[PromotionPolicy]:
        """获取适用的优惠政策"""
        applicable = []
        
        for policy in self.policies.get(provider_id, []):
            if self._check_policy_conditions(policy, usage):
                applicable.append(policy)
        
        return applicable
    
    def _check_policy_conditions(self, policy: PromotionPolicy, usage: Dict) -> bool:
        """检查政策条件"""
        # 检查时间有效性
        now = datetime.now()
        if policy.end_time and now > policy.end_time:
            return False
        
        # 检查用量条件
        conditions = policy.conditions
        if "min_tokens" in conditions and usage.get("total_tokens", 0) < conditions["min_tokens"]:
            return False
        
        if "max_tokens" in conditions and usage.get("total_tokens", 0) > conditions["max_tokens"]:
            return False
        
        return True
    
    def calculate_effective_cost(self, provider_id: str, base_cost: float, usage: Dict) -> float:
        """计算实际成本（考虑优惠）"""
        applicable_policies = self.get_applicable_policies(provider_id, usage)
        
        effective_cost = base_cost
        
        for policy in applicable_policies:
            if "discount_rate" in policy.benefits:
                effective_cost *= policy.benefits["discount_rate"]
        
        return effective_cost
```
