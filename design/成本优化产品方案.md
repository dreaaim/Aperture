# Aperture 模型网关成本优化产品方案

## 1. 产品背景与核心痛点

### 1.1 当前挑战
Aperture作为多模型网关，在实现智能路由的同时，面临着以下成本管理挑战：

1. **多服务商管理复杂性**：需要同时管理OpenAI、Claude、Gemini等多个服务商，每个服务商有不同的API接口、认证方式和计费规则
2. **模型能力与成本不匹配**：不同模型在能力、价格、响应速度上存在显著差异，需要根据任务难度智能分配
3. **计费模式多样化**：按token计费、按调用次数计费、按时间计费等不同计费模式并存
4. **配额管理困难**：需要实时监控各模型的配额使用情况，避免超出配额导致服务中断
5. **性能与成本平衡**：在保证响应速度的同时，需要选择成本最优的模型

### 1.2 现有基础分析
基于对当前代码架构的分析，Aperture已具备以下基础能力：

- **智能路由框架**：具备意图分类、难度评估、模型评分等核心路由能力
- **监控服务**：已实现基本的模型使用统计和性能监控
- **缓存机制**：支持语义缓存和Few-shot学习，减少重复调用
- **故障容错**：具备熔断机制和健康检查

## 2. 产品愿景与目标

### 2.1 产品愿景
构建业界最智能、最经济的多模型网关，实现"按需分配、成本最优、体验最佳"的智能化模型路由。

- **动态成本控制**：实现基于实时价格和配额的成本优化

#### 性能目标
- **响应时间保证**：95%的请求响应时间控制在3秒以内
- **可用性保证**：系统整体可用性达到99.9%

```python
class CostOptimizedRouter:
    def calculate_model_score(self, model: ModelStatus, intent: str, complexity: float) -> float:
        """计算模型的综合评分"""
        
        # 成本因子（权重：0.4）
        cost_factor = self._calculate_cost_factor(model)
        
        # 性能因子（权重：0.3）
        performance_factor = self._calculate_performance_factor(model)
        
        # 能力匹配因子（权重：0.2）
        capability_factor = self._calculate_capability_factor(model, intent, complexity)
        
        # 配额健康度因子（权重：0.1）
        quota_factor = self._calculate_quota_factor(model)
        
        # 综合评分
        total_score = (0.4 * cost_factor + 
                      0.3 * performance_factor + 
                      0.2 * capability_factor + 
                      0.1 * quota_factor)
        
        return total_score
```

#### 3.1.2 成本因子计算
```python
def _calculate_cost_factor(self, model: ModelStatus) -> float:
    """计算成本因子（成本越低，分数越高）"""
    
    # 获取当前实时价格（考虑促销活动）
    current_price = self._get_real_time_price(model.model_id)
    
    # 归一化处理：将价格映射到0-1区间
    max_price = self._get_max_price_in_catalog()
    normalized_price = current_price / max_price
    
    # 成本因子 = 1 - 归一化价格
    cost_factor = 1.0 - normalized_price
    
    return max(0.0, min(1.0, cost_factor))
```

```python
class QuotaManager:
    def __init__(self):
        self.quota_usage = {}  # 模型ID -> 已使用配额
        self.quota_limits = {}  # 模型ID -> 配额上限
        self.alert_threshold = 0.8  # 预警阈值（80%）
    
    def check_quota_health(self, model_id: str) -> dict:
        """检查配额健康度"""
        used = self.quota_usage.get(model_id, 0)
        limit = self.quota_limits.get(model_id, float('inf'))
        
        usage_ratio = used / limit if limit > 0 else 0
        
        return {
            "model_id": model_id,
            "used": used,
            "limit": limit,
            "usage_ratio": usage_ratio,
            "status": "healthy" if usage_ratio < self.alert_threshold else "warning",
            "remaining": limit - used
        }
```

```python
def calculate_cache_benefit(self, query: str, model_cost: float) -> float:
    """计算缓存命中的成本效益"""
    
    # 缓存命中节省的成本
    cost_saving = model_cost
    
    # 缓存维护成本（存储、计算等）
    cache_maintenance_cost = self._estimate_cache_cost()
    
    # 净效益
    net_benefit = cost_saving - cache_maintenance_cost
    
    return max(0, net_benefit)
```

```python
def should_switch_provider(self, current_model: str, alternative_model: str) -> bool:
    """判断是否应该切换服务商"""
    
    # 价格差异阈值（10%以上差异才考虑切换）
    price_difference_threshold = 0.1
    
    current_price = self._get_model_price(current_model)
    alternative_price = self._get_model_price(alternative_model)
    
    price_ratio = alternative_price / current_price
    
    # 只有当价格差异显著且性能影响可接受时才切换
    if price_ratio < (1 - price_difference_threshold):
        performance_impact = self._estimate_performance_impact(current_model, alternative_model)
        if performance_impact < self.performance_threshold:
            return True
    
    return False
```

#### 4.2.1 成本优化引擎（CostOptimizationEngine）
```python
class CostOptimizationEngine:
    """成本优化核心引擎"""
    
    def __init__(self):
        self.router = CostOptimizedRouter()
        self.quota_manager = QuotaManager()
        self.price_monitor = PriceMonitor()
        self.cache_manager = CacheManager()
    
    async def select_optimal_model(self, query: str, user_id: str = None) -> ModelSelection:
        """选择最优模型"""
        
        # 1. 意图分类和难度评估
        intent_info = await self._classify_intent(query)
        
        # 2. 检查缓存命中
        cache_result = await self.cache_manager.check_cache(query, intent_info)
        if cache_result.hit:
            return ModelSelection(
                model_id="cache",
                cache_hit=True,
                cost_saving=cache_result.cost_saving
            )
        
        # 3. 获取候选模型
        candidate_models = self._get_candidate_models(intent_info)
        
        # 4. 计算各模型评分
        scored_models = []
        for model in candidate_models:
            score = self.router.calculate_model_score(model, intent_info)
            scored_models.append((model, score))
        
        # 5. 选择最优模型
        best_model = max(scored_models, key=lambda x: x[1])[0]
        
        return ModelSelection(
            model_id=best_model.model_id,
            cost_estimate=self._estimate_cost(best_model, query),
            performance_estimate=self._estimate_performance(best_model)
        )
```

```python
class PriceMonitor:
    """实时价格监控器"""
    
    def __init__(self):
        self.price_cache = {}
        self.cache_ttl = 300  # 5分钟缓存
        self.price_sources = [
            OpenAIPriceSource(),
            ClaudePriceSource(),
            GeminiPriceSource()
        ]
    
    async def get_current_price(self, model_id: str) -> float:
        """获取当前实时价格"""
        
        # 检查缓存
        if model_id in self.price_cache:
            cached_price, timestamp = self.price_cache[model_id]
            if time.time() - timestamp < self.cache_ttl:
                return cached_price
        
        # 从多个源获取价格
        prices = []
        for source in self.price_sources:
            try:
                price = await source.get_price(model_id)
                if price is not None:
                    prices.append(price)
            except Exception as e:
                logger.warning(f"Failed to get price from {source}: {e}")
        
        # 取最低价格
        current_price = min(prices) if prices else self._get_default_price(model_id)
        
        # 更新缓存
        self.price_cache[model_id] = (current_price, time.time())
        
        return current_price
```

```python
def calculate_roi(self, implementation_cost: float, monthly_savings: float) -> dict:
    """计算投资回报率"""
    
    payback_period = implementation_cost / monthly_savings  # 回本周期（月）
    annual_roi = (monthly_savings * 12 - implementation_cost) / implementation_cost
    
    return {
        "implementation_cost": implementation_cost,
        "monthly_savings": monthly_savings,
        "payback_period_months": payback_period,
        "annual_roi_percentage": annual_roi * 100
    }
```
